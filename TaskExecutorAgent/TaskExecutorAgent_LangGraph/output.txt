#### Code for `Services/TodoService.cs`
```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using Models;

namespace Services
{
    public class TodoService
    {
        private readonly List<TodoItem> _todoItems = new List<TodoItem>();
        private int _nextId = 1;

        public TodoItem Add(string title)
        {
            if (string.IsNullOrWhiteSpace(title) || title.Length > 100)
            {
                throw new ArgumentException("Title must be non-empty and less than or equal to 100 characters.");
            }

            var todoItem = new TodoItem(_nextId++, title);
            _todoItems.Add(todoItem);
            return todoItem;
        }

        public bool MarkDone(int id)
        {
            var item = _todoItems.FirstOrDefault(t => t.Id == id);
            if (item == null) return false;

            item.IsDone = true;
            return true;
        }

        public IReadOnlyList<TodoItem> GetAll()
        {
            return _todoItems.AsReadOnly();
        }
    }
}
```
